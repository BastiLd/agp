        const contentWrapper = document.getElementById('content-wrapper');
        const pageTitleElement = document.getElementById('page-title');
        const mobileMenu = document.getElementById('mobile-menu');
        const burgerIcon = document.getElementById('burger-icon');

        /**
         * Lädt den Inhalt basierend auf dem Seiten-Key dynamisch.
         * @param {string} pageKey - Der Schlüssel der zu ladenden Seite ('home', 'kindersicherheit' etc.).
         */
        function loadPage(pageKey) {
            const pageData = PAGES[pageKey] || PAGES['home']; // Fallback zu Home
            
            // Inhalt mit Fade-In-Animation aktualisieren
            contentWrapper.innerHTML = `<div class="subpage-content fade-in">${pageData.content}</div>`;
            pageTitleElement.textContent = pageData.title;

            // URL-Hash aktualisieren, um die Historie zu pflegen
            history.pushState(null, null, `#${pageKey}`);

            // Aktiven Link in der Navigation hervorheben
            updateActiveLink(pageKey);
        }

        /**
         * Hebt den aktuell aktiven Navigationslink hervor.
         * @param {string} activeKey - Der aktive Seiten-Schlüssel.
         */
        function updateActiveLink(activeKey) {
            document.querySelectorAll('.nav-links a').forEach(link => {
                const linkHash = link.getAttribute('href').substring(1);
                if (linkHash === activeKey) {
                    link.classList.add('active');
                    link.classList.add('text-[var(--primary-color)]');
                    link.classList.remove('text-white');
                } else {
                    link.classList.remove('active');
                    link.classList.remove('text-[var(--primary-color)]');
                    link.classList.add('text-white');
                }
            });
        }

        /**
         * Schaltet das mobile Navigationsmenü um.
         */
        function toggleMenu() {
            const isHidden = mobileMenu.classList.toggle('hidden-mobile');
            mobileMenu.classList.toggle('visible-mobile', !isHidden);
            burgerIcon.classList.toggle('fa-times', !isHidden); // Icon zu X wechseln, wenn geöffnet
            burgerIcon.classList.toggle('fa-bars', isHidden); // Icon zu Burger wechseln, wenn geschlossen
            burgerIcon.classList.toggle('open', !isHidden);
        }

        /**
         * Initialisiert die Seite beim Laden oder Navigieren über den Browser-Zurück-Button.
         */
        function init() {
            // Aktuellen Hash aus der URL abrufen
            const currentHash = window.location.hash.substring(1) || 'home';
            loadPage(currentHash);

            // Listener für den Browser-Zurück-Button hinzufügen
            window.addEventListener('popstate', () => {
                const popHash = window.location.hash.substring(1) || 'home';
                loadPage(popHash);
            });
        }
